#   pip uninstall -y torch
#   pip install torch==2.14.1 torchvision==0.29.1 --index-url https://download.pytorch.org/whl/cu130

# torchvision is required, not optional: Ministral 3 uses Pixtral's vision stack and
# PixtralProcessor refuses to load without it. Install it from the same CUDA index as torch,
# or pip will pull a CPU build and quietly mismatch the two.
